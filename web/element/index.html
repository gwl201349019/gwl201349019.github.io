# MessageBox 审批操作

TIP

业务审批代码相关 textarea类型 1同意 2拒绝
status状态2为必填 (/.+/)

  let info = status === 1 ? '请说明同意原因（选填）' : '请说明拒绝原因'

  this.$prompt(info, {
    closeOnClickModal: false,
    confirmButtonText: '确定',
    cancelButtonText: '取消',
    inputErrorMessage: '拒绝必须填理由',
    inputPattern: status === 1 ? (/.*/) : (/.+/),
    inputType: 'textarea'
  }).then(({ value }) => {
    let params = {}
    this.post('api', params)
      .then((response) => {
        this.$message({
          type: 'success',
          message: response.msg
        })
      })
      .catch((error) => {
        this.$message({
          showClose: true,
          message: error,
          type: 'error'
        })
      })
  })

# 下拉框 allow-create 离开焦点保存

 <el-select
    id="selectId"
    v-model="selectValue"
    placeholder=""
    clearable
    filterable
    allow-create
    style="width: 100%"
    @blur="blurChange"
    @visible-change="visibleChange">
  <el-option v-for="item in list" :key="item.id" :label="item.name" :value="item.id"> </el-option>
</el-select>

const blurChange = (e) => {
  const value = e.target.value
  if (value) {
    if (selectValue.indexOf(value) > -1 && list.value.some((row) => row.name == selectValue)) {
      submit()
    } else {
      selectValue = value
      submit()
    }
  }
}
